    return PaginatedPostSearch(

@router.post("", status_code=status.HTTP_201_CREATED, response_model=ItemResponseMini)
async def create_post(
    item_data: ItemCreate,
    db: Annotated[AsyncSession, Depends(get_db)],
    current_user: CurrentUser,
):
    result = await db.execute(
        select(models.Item).where(
            func.lower(models.Item.title) == item_data.title.lower(),
            models.Item.user_id == current_user.id,
        )
    )

    existing_item = result.scalars().first()

    if existing_item:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Item with same title already exists",
        )

    new_item = models.Item(
        title=item_data.title,
        content=item_data.content,
        is_public=item_data.is_public,
        user_id=current_user.id,
    )

    if item_data.tag_ids:

        result = await db.execute(
            select(models.Tag).where(models.Tag.id.in_(item_data.tag_ids))
        )

        tags = result.scalars().all()

        if len(tags) != len(item_data.tag_ids):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST, detail="some tags are invalid"
            )

        new_item.tags = tags

    db.add(new_item)
    await db.commit()

    #adding the frist version

    version = models.Version(
        item_id = new_item.id,
        version_number = 1,
        title = new_item.title,
        content = new_item.content

    )

    db.add(version)
    await db.commit()


    # pre-loading the tags for return object
    result = await db.execute(
        select(models.Item)
        .options(selectinload(models.Item.tags))
        .where(models.Item.id == new_item.id)
    )

    item_with_tags = result.scalars().first()

    return item_with_tags

@router.get("/{item_id}", response_model=ItemResponse)
async def get_item_by_id(item_id: int, db: Annotated[AsyncSession, Depends(get_db)]):

    result = await db.execute(
        select(models.Item)
        .options(selectinload(models.Item.tags))
        .where(models.Item.id == item_id)
    )

    item = result.scalars().first()

    if not item:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST, detail="Item dosent exists"
        )

    if item.is_public == False:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Not authorized to access the post",
        )

    # the counts

    likes_count = await db.scalar(
        select(func.count(models.Like.item_id)).where(models.Like.item_id == item_id)
    )

    comments_count = await db.scalar(
        select(func.count(models.Comment.id)).where(models.Comment.item_id == item_id)
    )

    bookmarks_count = await db.scalar(
        select(func.count(models.Bookmark.item_id)).where(
            models.Bookmark.item_id == item_id
        )
    )

    versions_count = await db.scalar(
        select(func.count(models.Version.id)).where(models.Version.item_id == item_id)
    )

    # fetching the latest version

    result = await db.execute(
        select(models.Version)
        .where(models.Version.item_id == item_id)
        .order_by(models.Version.version_number.desc())
        .limit(1)
    )

    latest_version = result.scalars().first()

    return {
        "id": item_id,
        "user_id" : item.user_id,
        "title": item.title,
        "content": item.content,
        "is_public": item.is_public,
        "created_at": item.created_at,
        "tags": item.tags,
        "likes_count": likes_count,
        "comments_count": comments_count,
        "version_count": versions_count,
        "latest_version": latest_version,
    }

@router.patch("/{item_id}",
            status_code= status.HTTP_200_OK,
            response_model= ItemUpdateResponse
        )
async def update_item_version(item_id : int, data: ItemUpdate, db: Annotated[AsyncSession, Depends(get_db)], current_user : CurrentUser):

    result = await db.execute(select(models.Item).where(models.Item.id == item_id))

    item = result.scalars().first()

    if not item:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Item not found",
        )
    
    if item.user_id != current_user.id:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Not authorized to access the Item",
        )
    
    latest_version = await db.scalar(
        select(func.max(models.Version.version_number))
               .where(models.Version.item_id == item_id)
    )

    if data.content == item.content and data.title == item.title :
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Can't update the item with same content or title",
        )
    
    updated_title = data.title if data.title else item.title
    updated_content = data.content if data.content else item.content
    new_version = models.Version(
        item_id = item_id,
        version_number = latest_version + 1,
        title = updated_title,
        content = updated_content

    )

    item.title =  updated_title
    item.content = updated_content

    db.add(new_version)
    await db.commit()
    await db.refresh(new_version)

   

    return new_version

from fastapi import APIRouter, Depends, Query
from typing import Annotated, Optional
from sqlalchemy import select, func, case
from sqlalchemy.orm import selectinload
from sqlalchemy.ext.asyncio import AsyncSession
from database import get_db
from schemas import PaginatedFeedResponse
from auth import get_current_user_optional
import models
from config import settings

router = APIRouter(prefix="/api/feed")

@router.get("", response_model=PaginatedFeedResponse)
async def get_feed(
    db: Annotated[AsyncSession,Depends(get_db)],
    skip : Annotated[int, Query(ge=0)] = 0,
    limit : Annotated[int, Query(ge=1, le=100)] = settings.post_per_page,
    current_user: Optional[models.User] = Depends(get_current_user_optional)
):

    count_result = await db.execute(select(func.count()).select_from(models.Item).where(models.Item.is_public==True))
    total = count_result.scalar() or 0

    if current_user:

        followed_results = await db.execute(
            select(models.Follow.following_id)
            .where(models.Follow.follower_id == current_user.id)
        )

        followed_user_ids = [row[0] for row in followed_results.all()]

        ranking_case =  case(
            (models.Item.user_id.in_(followed_user_ids),1),
            (models.Item.user_id == current_user.id, 2),
            else_ = 3
        )


        query = (
            select(models.Item)
            .options(selectinload(models.Item.tags))
            .where(models.Item.is_public == True)
            .order_by(ranking_case, models.Item.created_at.desc())
        )

    else:

        query = (
            select(models.Item)
            .options(selectinload(models.Item.tags))
            .where(models.Item.is_public == True)
            .order_by(models.Item.created_at.desc())
        )

    result = await db.execute(
        query.offset(skip).limit(limit)
    )

    items = result.scalars().all()
    
    if not items:

        return PaginatedFeedResponse(
            items = [],
            total = total,
            skip = skip,
            limit = limit,
            has_more = False
        )
    
    item_ids = [item.id for item in items]

    likes_result = await db.execute( select(models.Like.item_id, func.count())
                                    .where(models.Like.item_id.in_(item_ids))
                                    .group_by(models.Like.item_id)
)

    likes_count = dict(likes_result.all())

    comments_result = await db.execute(select(models.Comment.item_id, func.count(models.Comment.id))
                                       .where(models.Comment.item_id.in_(item_ids))
                                       .group_by(models.Comment.item_id)
                                       )
    
    comments_count = dict(comments_result.all())

    bookmarks_result = await db.execute(select(models.Bookmark.item_id, func.count(models.Bookmark.item_id))
                                        .where(models.Bookmark.item_id.in_(item_ids))
                                        .group_by(models.Bookmark.item_id)
                                        )
    
    bookmarks_count = dict(bookmarks_result.all())

    versions_result = await db.execute(select(models.Version.item_id, func.count(models.Version.id))
                                       .where(models.Version.item_id.in_(item_ids))
                                       .group_by(models.Version.item_id)
                                       )
    
    versions_count = dict(versions_result.all())

    feed=[]

    for item in items:

        feed.append({
            "id" : item.id,
            "user_id" : item.user_id,
            "title" : item.title,
            "content" : item.content,
            "is_public" : item.is_public,
            "created_at" : item.created_at,
            "tags" : item.tags,

            "likes_count" : likes_count.get(item.id, 0),
            "comments_count" : comments_count.get(item.id, 0),
            "bookmarks_count" : bookmarks_count.get(item.id, 0),
            "versions_count" : versions_count.get(item.id, 0)
        })

    has_more = skip + len(items) < total


    return PaginatedFeedResponse(
        items = feed,
        total = total,
        skip = skip,
        limit = limit,
        has_more = has_more
    )
